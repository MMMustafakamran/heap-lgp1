import { Image, StyleSheet, Text, View } from "react-native";
import { z } from "zod";
import {
  createGenerativeUI,
  type GenerativeUIComponentProps,
} from "@copilotkit/react-native/generative-ui";

function ProductGrid({ title, children }: GenerativeUIComponentProps<typeof shopUI, "ProductGrid">) {
  return (
    <View>
      {title ? <Text style={styles.title}>{title}</Text> : null}
      <View style={styles.grid}>{children}</View>
    </View>
  );
}

function ProductCard({ name, price, imageUrl }: GenerativeUIComponentProps<typeof shopUI, "ProductCard">) {
  return (
    <View style={styles.card}>
      <Image source={{ uri: imageUrl }} style={styles.image} accessibilityLabel={name} />
      <Text style={styles.name}>{name}</Text>
      <Text>${price}</Text>
    </View>
  );
}

export const shopUI = createGenerativeUI({
  id: "acme/shop@1",
  components: {
    ProductGrid: {
      component: ProductGrid,
      description: "A grid of products. Put the ids of ProductCard components in children.",
      props: z.object({ title: z.string().optional() }),
      slots: { children: "multiple" },
    },
    ProductCard: {
      component: ProductCard,
      description: "One product. Copy every value from a product of a tool result.",
      props: z.object({ name: z.string(), price: z.number(), imageUrl: z.string() }),
    },
  },
});

const styles = StyleSheet.create({
  title: { fontSize: 17, fontWeight: "600", marginBottom: 8 },
  grid: { flexDirection: "row", flexWrap: "wrap" },
  // A fixed width, not a percentage: when the agent sends an `accessibility`
  // envelope, the card sits in a wrapper View, and a percentage would resolve
  // against that wrapper rather than the grid (see below).
  // Margins rather than `gap`, which needs React Native 0.71.
  card: { width: 160, marginRight: 12, marginBottom: 12 },
  image: { aspectRatio: 1, borderRadius: 8 },
  name: { fontWeight: "600", marginTop: 4 },
});

### [App.tsx]
import { Platform } from "react-native";
import { CopilotKitProvider } from "@copilotkit/react-native/headless";
import { GenerativeUIProvider } from "@copilotkit/react-native/generative-ui";
import { shopUI } from "./src/shopUI";
import { ShopScreen } from "./src/ShopScreen";

// The Android emulator reaches your machine at 10.0.2.2, not localhost.
const runtimeUrl =
  Platform.OS === "android"
    ? "http://10.0.2.2:8200/api/copilotkit"
    : "http://localhost:8200/api/copilotkit";

export default function App() {
  return (
    <CopilotKitProvider runtimeUrl={runtimeUrl}>
      <GenerativeUIProvider ui={shopUI}>
        <ShopScreen />
      </GenerativeUIProvider>
    </CopilotKitProvider>
  );
}
